import React, { useMemo, useEffect } from "react";
import { View, StyleSheet, Text, Dimensions } from "react-native";
import Animated, { 
  useSharedValue, 
  useAnimatedStyle, 
  withTiming,
  withRepeat,
  cancelAnimation,
  useAnimatedProps
} from "react-native-reanimated";
import { Gesture, GestureDetector } from "react-native-gesture-handler";
import Svg, { Defs, Filter, FeGaussianBlur, Polyline, Circle } from "react-native-svg";

import FloorMap from "../../assets/maps/floor_1.svg";
import { NODES, ROOMS, CORRIDORS } from "../../utils/constants";

const { width: SCREEN_WIDTH, height: SCREEN_HEIGHT } = Dimensions.get("window");
const MAP_SIZE = 1200;

// High-performance native components wrappers
const AnimatedPolyline = Animated.createAnimatedComponent(Polyline);
const AnimatedCircle = Animated.createAnimatedComponent(Circle);

export default function IndoorDeptMap({ startRoom, endRoom, route, setStartRoom, setEndRoom }) {
  
  // ================= CAMERA & GESTURES (YOUR ORIGINAL EXACT MECHANISM) =================
  const scale = useSharedValue(1);
  const translateX = useSharedValue(0);
  const translateY = useSharedValue(0);
  
  const startX = useSharedValue(0);
  const startY = useSharedValue(0);
  const startScale = useSharedValue(1);

  // Performance-optimized Shared Values (No heavy bridge traffic)
  const strokeOffset = useSharedValue(2000); // Start fully hidden
  const pulseScale = useSharedValue(1);

  // 1. Loop pulse animation runs on UI Thread directly
  useEffect(() => {
    pulseScale.value = withRepeat(withTiming(1.6, { duration: 1000 }), -1, true);
  }, []);

  // 2. Trigger route animation only when target change occurs
  useEffect(() => {
    if (route.length >= 2) {
      strokeOffset.value = 2000; // Reset
      strokeOffset.value = withTiming(0, { duration: 800 }); // Animate to visible smoothly
    }
  }, [route]);

  const panGesture = Gesture.Pan()
    .onBegin(() => {
      cancelAnimation(translateX);
      cancelAnimation(translateY);
      startX.value = translateX.value;
      startY.value = translateY.value;
    })
    .onUpdate((e) => {
      translateX.value = startX.value + e.translationX;
      translateY.value = startY.value + e.translationY;
    });

  const pinchGesture = Gesture.Pinch()
    .onBegin(() => {
      cancelAnimation(scale);
      startScale.value = scale.value;
    })
    .onUpdate((e) => {
      scale.value = Math.max(0.6, Math.min(startScale.value * e.scale, 4));
    });

  const doubleTap = Gesture.Tap().numberOfTaps(2).onEnd(() => {
    scale.value = withTiming(1, { duration: 250 });
    translateX.value = withTiming(0, { duration: 250 });
    translateY.value = withTiming(0, { duration: 250 });
  });

  const gesture = Gesture.Simultaneous(panGesture, pinchGesture, doubleTap);

  const focusRoom = (room) => {
    // Camera jump logic removed to completely avoid layout stuttering
  };

  // ================= CACHED ROUTE & PATH STRING =================
  const { pointsString, firstNode, lastNode } = useMemo(() => {
    if (route.length < 2) return { pointsString: "", firstNode: null, lastNode: null };
    
    const validNodes = route.map((id) => NODES.find((n) => n.id === id)).filter(Boolean);
    if (validNodes.length === 0) return { pointsString: "", firstNode: null, lastNode: null };

    return {
      pointsString: validNodes.map((p) => `${p.x},${p.y}`).join(" "),
      firstNode: validNodes[0],
      lastNode: validNodes[validNodes.length - 1],
    };
  }, [route]);

  // ================= NATIVE ANIMATED PROPS (NO RE-RENDERS) =================
  const animatedPathProps = useAnimatedProps(() => ({
    strokeDasharray: "2000",
    strokeDashoffset: strokeOffset.value
  }));

  const animatedPulseProps = useAnimatedProps(() => ({
    r: 8 * pulseScale.value,
    opacity: 1 - (pulseScale.value - 1) / 0.6
  }));

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: translateX.value },
      { translateY: translateY.value },
      { scale: scale.value }
    ],
  }));

  return (
    <GestureDetector gesture={gesture}>
      <View style={styles.container}>
        <Animated.View style={[styles.mapWrapper, animatedStyle]}>
          
          {/* Base Floor Map */}
          <FloorMap width={MAP_SIZE} height={MAP_SIZE} />

          {/* Static Corridors */}
          <Svg width={MAP_SIZE} height={MAP_SIZE} style={styles.absoluteSvg}>
            {CORRIDORS.map((c) => (
              <Polyline 
                key={c.id} 
                points={c.points.map(p => `${p.x},${p.y}`).join(" ")} 
                stroke="#2A2F45" 
                strokeWidth={4} 
                fill="none" 
              />
            ))}
          </Svg>

          {/* Active Route Line Layers */}
          {pointsString.length > 0 && (
            <Svg width={MAP_SIZE} height={MAP_SIZE} style={styles.absoluteSvg}>
              <Defs>
                <Filter id="innerGlow" x="-20%" y="-20%" width="140%" height="140%">
                  <FeGaussianBlur stdDeviation="3" result="blur" />
                </Filter>
                <Filter id="outerGlow" x="-50%" y="-50%" width="200%" height="200%">
                  <FeGaussianBlur stdDeviation="8" result="blur" />
                </Filter>
              </Defs>

              {/* LAYER 1: Deep Outer Glow */}
              <AnimatedPolyline
                points={pointsString}
                stroke="#5755EA"
                strokeWidth={14}
                opacity={0.3}
                fill="none"
                strokeLinecap="round"
                strokeLinejoin="round"
                filter="url(#outerGlow)"
                animatedProps={animatedPathProps}
              />

              {/* LAYER 2: Soft Inner Glow */}
              <AnimatedPolyline
                points={pointsString}
                stroke="#5755EA"
                strokeWidth={8}
                opacity={0.6}
                fill="none"
                strokeLinecap="round"
                strokeLinejoin="round"
                filter="url(#innerGlow)"
                animatedProps={animatedPathProps}
              />

              {/* LAYER 3: Main Core Line */}
              <AnimatedPolyline
                points={pointsString}
                stroke="#9795FF"
                strokeWidth={3.5}
                opacity={1}
                fill="none"
                strokeLinecap="round"
                strokeLinejoin="round"
                animatedProps={animatedPathProps}
              />

              {/* STARTPOINT PIN (Exact Node Entrance Position) */}
              {firstNode && (
                <Circle
                  cx={firstNode.x}
                  cy={firstNode.y}
                  r={4.5}
                  fill="#635BFF"
                  filter="url(#innerGlow)"
                />
              )}

              {/* DESTINATION NEON PULSE DOT (Far end near target corridor node) */}
              {lastNode && (
                <>
                  <AnimatedCircle
                    cx={lastNode.x}
                    cy={lastNode.y}
                    fill="none"
                    stroke="#00F5D4"
                    strokeWidth={2}
                    animatedProps={animatedPulseProps}
                  />
                  <Circle
                    cx={lastNode.x}
                    cy={lastNode.y}
                    r={4.5}
                    fill="#00F5D4"
                    filter="url(#innerGlow)"
                  />
                </>
              )}
            </Svg>
          )}

          {/* YOUR ORIGINAL ROOM LABELS & DESIGN (Untouched positions) */}
          {ROOMS.map((room) => {
            const isSelected = startRoom === room.id || endRoom === room.id;
            return (
              <View
                key={room.id}
                style={[
                  styles.roomPin,
                  { 
                    left: room.x, 
                    top: room.y, 
                    backgroundColor: isSelected ? "#00F5D4" : "#635BFF", 
                    transform: [{ scale: isSelected ? 1.15 : 1 }, { translateX: -25 }, { translateY: -10 }] 
                  }
                ]}
              >
                <Text
                  onPress={() => {
                    if (!startRoom) {
                      setStartRoom(room.id);
                    } else if (startRoom && !endRoom) {
                      setEndRoom(room.id);
                    } else {
                      setStartRoom(room.id);
                      setEndRoom(null);
                    }
                    focusRoom(room);
                  }}
                  style={styles.roomText}
                >
                  {room.name}
                </Text>
              </View>
            );
          })}
        </Animated.View>
      </View>
    </GestureDetector>
  );
}

const styles = StyleSheet.create({
  container: { 
    flex: 1, 
    backgroundColor: "#050816",
    overflow: "hidden" 
  },
  mapWrapper: {
    width: MAP_SIZE,
    height: MAP_SIZE,
  },
  absoluteSvg: {
    position: "absolute",
    top: 0,
    left: 0,
  },
  roomPin: { 
    position: "absolute", 
    paddingHorizontal: 6, 
    paddingVertical: 4, 
    borderRadius: 6, 
    zIndex: 10, 
    elevation: 5,
  },
  roomText: {
